293  double latLon_to_XY(double ref_latitude, double ref_longitude, double given_latitude, double given_longitude,
294  double enu_to_asset_frame_yaw,
295  double &X, double &Y, double earth_radius)
296  {
297  double REFERENCE_LAT = M_PI * ref_latitude / 180.0;
298  double REFERENCE_LONG = M_PI * (ref_longitude) / 180.0;
299  const double EARTH = earth_radius;
300  const double ENU_TO_ENU_YAW_RAD = M_PI * enu_to_asset_frame_yaw / 180.0;
301 
302  const double lat_radians = M_PI * given_latitude / 180.0;
303  const double long_radians = M_PI * given_longitude / 180.0;
304  const double radians_delta_latitude = lat_radians - REFERENCE_LAT;
305  const double radians_delta_longitude = long_radians - REFERENCE_LONG;
306 
307  double a = sin(radians_delta_latitude / 2) * sin(radians_delta_latitude / 2) +
308  cos(REFERENCE_LAT) * cos(lat_radians) * sin(radians_delta_longitude / 2) * sin(radians_delta_longitude / 2);
309  double c = 2 * atan2(sqrt(a), sqrt(1 - a));
310  double d = c * EARTH;
311  double bearing = atan2(sin(long_radians - REFERENCE_LONG) * cos(lat_radians),
312  cos(REFERENCE_LAT) * sin(lat_radians) - sin(REFERENCE_LAT) * cos(lat_radians) * cos(long_radians - REFERENCE_LONG));
313  X = cos(bearing - ENU_TO_ENU_YAW_RAD) * d;
314  Y = -sin(bearing - ENU_TO_ENU_YAW_RAD) * d;
315 
316  return bearing;
317  }

861  bool utmToLatLon(double utm_easting, double utm_northing, const std::string &utm_zone,
862  double &latitude, double &longitude)
863  {
864  if (utm_zone.empty())
865  {
866  return false;
867  }
868 
869  //
870  // 1. Parse UTM zone ("10N", "33T", "14S", or just "10")
871  //
872  size_t i = 0;
873  while (i < utm_zone.size() && std::isdigit(static_cast<unsigned char>(utm_zone[i])))
874  {
875  ++i;
876  }
877 
878  int zone_number = std::stoi(utm_zone.substr(0, i));
879 
880  bool northern = true; // default
881  if (i < utm_zone.size())
882  {
883  char letter = std::toupper(utm_zone[i]);
884  northern = (letter >= 'N'); // C–M = south, N–X = north
885  }
886 
887  //
888  // 2. WGS84 parameters
889  //
890  const double a = 6378137.0;
891  const double eccSquared = 0.00669437999014;
892  const double k0 = 0.9996;
893 
894  //
895  // 3. Remove false Easting/Northing
896  //
897  double x = utm_easting - 500000.0;
898  double y = utm_northing;
899 
900  if (!northern)
901  y -= 10000000.0; // 10,000,000 m offset for southern hemisphere
902 
903  //
904  // 4. Compute footpoint latitude
905  //
906  double eccPrimeSquared = eccSquared / (1.0 - eccSquared);
907 
908  double M = y / k0;
909  double mu = M / (a * (1.0 - eccSquared / 4.0 -
910  3.0 * std::pow(eccSquared, 2) / 64.0 -
911  5.0 * std::pow(eccSquared, 3) / 256.0));
912 
913  double e1 = (1.0 - std::sqrt(1.0 - eccSquared)) /
914  (1.0 + std::sqrt(1.0 - eccSquared));
915 
916  double J1 = (3.0 * e1 / 2.0 - 27.0 * std::pow(e1, 3) / 32.0);
917  double J2 = (21.0 * std::pow(e1, 2) / 16.0 - 55.0 * std::pow(e1, 4) / 32.0);
918  double J3 = (151.0 * std::pow(e1, 3) / 96.0);
919  double J4 = (1097.0 * std::pow(e1, 4) / 512.0);
920 
921  double phi1 = mu + J1 * std::sin(2.0 * mu) + J2 * std::sin(4.0 * mu) + J3 * std::sin(6.0 * mu) + J4 * std::sin(8.0 * mu);
922 
923  //
924  // 5. Compute latitude and longitude
925  //
926  double sinPhi1 = std::sin(phi1);
