154  pcl::PointCloud<pcl::PointXYZ> &cloud, geometry_msgs::msg::TransformStamped &transform, std::string &error_msg,

156  {

159  {
160  cv_ptr = cv_bridge::toCvCopy(image, sensor_msgs::image_encodings::TYPE_32FC1);
161  }

163  {
164  error_msg = std::string(e.what());
165  return false;
166  }
167 
168  cv::Mat frame = cv_ptr->image;
169 
170  // get camera intrinsics
171  float fx = cam_info->k[0];
172  float fy = cam_info->k[4];
173  float cx = cam_info->k[2];
174  float cy = cam_info->k[5];
175 
176  pcl::PointXYZ bad_point;
177  ros2::PCLUtils::setNaN(bad_point);
178 
179  // setup cloud
180  cloud.width = image->width / downsample_factor; // frame.cols

182 
183  // RCLCPP_INFO(node_->get_logger(), "cloud[%d, %d]", cloud.width, cloud.height);
184 
185  cloud.points.resize(cloud.width * cloud.height);
186  int cnt = -1;
187  for (int r = 0; r < frame.rows; r += downsample_factor)
188  {
189  for (int c = 0; c < frame.cols; c += downsample_factor)
190  {
191 
192  cnt++;
193 
194  // RCLCPP_INFO(node_->get_logger(), "r:%d c:%d, cnt: %d, cloud.points: %d", r,c, cnt, (int)cloud.points.size());
195  int idx = r * frame.cols + c;
196 
197  float z = frame.at<float>(r, c) / depth_factor;
198  // ROS_INFO("z: %f vs raw_z: %f", z, frame.at<float>(r,c));
199 
200  if (z == 0 || std::isnan(z) || z < min_distance || z > max_distance)
201  {
202  cloud.points[cnt] = bad_point;
203  continue;
204  }
205 
206  pcl::PointXYZ p;
207  p.z = z;
