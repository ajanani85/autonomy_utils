95  float q3q3 = q3 * q3;
96  float q4q4 = q4 * q4;
97 
98  // Outlier rejection for accelerometer
99  norm = std::sqrt(ax * ax + ay * ay + az * az);
100  if (norm < min_accel_norm_ || norm > accel_outlier_threshold_)
101  return; // reject outlier or free-fall
102  ax /= norm;
103  ay /= norm;
104  az /= norm;
105 
106  // Gradient descent algorithm corrective step
107  s1 = _4q1 * q3q3 + _2q3 * ax + _4q1 * q2q2 - _2q2 * ay;
108  s2 = _4q2 * q4q4 - _2q4 * ax + 4.0f * q1q1 * q2 - _2q1 * ay - _4q2 + _8q2 * q2q2 + _8q2 * q3q3 + _4q2 * az;
109  s3 = 4.0f * q1q1 * q3 + _2q1 * ax + _4q3 * q4q4 - _2q4 * ay - _4q3 + _8q3 * q2q2 + _8q3 * q3q3 + _4q3 * az;
110  s4 = 4.0f * q2q2 * q4 - _2q2 * ax + 4.0f * q3q3 * q4 - _2q3 * ay;
111  norm = std::sqrt(s1 * s1 + s2 * s2 + s3 * s3 + s4 * s4);
112  if (norm == 0.0f)
113  return;
114  s1 /= norm;
115  s2 /= norm;
116  s3 /= norm;
117  s4 /= norm;
118 
119  // Rate of change of quaternion
120  float qDot1 = 0.5f * (-q2 * gx - q3 * gy - q4 * gz) - beta_ * s1;
121  float qDot2 = 0.5f * (q1 * gx + q3 * gz - q4 * gy) - beta_ * s2;
122  float qDot3 = 0.5f * (q1 * gy - q2 * gz + q4 * gx) - beta_ * s3;
123  float qDot4 = 0.5f * (q1 * gz + q2 * gy - q3 * gx) - beta_ * s4;
124 
125  // Integrate to yield quaternion
126  q1 += qDot1 * samplePeriod_;
127  q2 += qDot2 * samplePeriod_;
128  q3 += qDot3 * samplePeriod_;
129  q4 += qDot4 * samplePeriod_;
130  norm = std::sqrt(q1 * q1 + q2 * q2 + q3 * q3 + q4 * q4);
131  q_.w = q1 / norm;
132  q_.x = q2 / norm;
133  q_.y = q3 / norm;
134  q_.z = q4 / norm;
135  }

168  float cy = std::cos(half_neg_yaw);

172  q_neg_yaw.w = cy;

174  q_neg_yaw.y = 0.0f;
175  q_neg_yaw.z = sy;
176  // Quaternion multiplication: q_no_yaw = q * q_neg_yaw
177  geometry_msgs::msg::Quaternion q_no_yaw;
178  q_no_yaw.w = q0 * q_neg_yaw.w - q1 * q_neg_yaw.x - q2 * q_neg_yaw.y - q3 * q_neg_yaw.z;
179  q_no_yaw.x = q0 * q_neg_yaw.x + q1 * q_neg_yaw.w + q2 * q_neg_yaw.z - q3 * q_neg_yaw.y;
180  q_no_yaw.y = q0 * q_neg_yaw.y - q1 * q_neg_yaw.z + q2 * q_neg_yaw.w + q3 * q_neg_yaw.x;
